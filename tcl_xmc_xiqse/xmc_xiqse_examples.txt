#The following script creates 100 VLANS with IP Addresses from 10.1.1.1/16 to 10.100.1.1/16:
enable cli scripting
Set var count 1
while ($count < 101) do
Create vlan v$count
configure vlan v$count ipaddress 10.$(count).1.1/16
set var count ($count + 1)
endwhile
show vlan

#The following script displays the date and time:
set var CLI.OUT " "
show switch
set var date $TCL(lrange ${CLI.OUT} 27 29)
set var year $TCL(lrange ${CLI.OUT} 31 31)
set var date $TCL(linsert $date 3 $year)
set var time $TCL(lrange ${CLI.OUT} 30 30)
show var date
show var time

#The following script sorts the FDB (forwarding database) table in descending order:
set var CLI.OUT " "
show fdb
set var x1 $TCL(split ${CLI.OUT} "\n")
set var x2 $TCL(lsort -decreasing $x1)
set var output $TCL(join $x2 "\n")
show var output

#The following script extracts the MAC address given the age of an FDB entry:
set var CLI.OUT " "
show fdb
set var input $TCL(split ${CLI.OUT} "\n")
set var y1 $TCL(lsearch -glob $input *age*)
set var y2 $TCL(lindex $input $y1)
set var y3 $TCL(split $y2 " ")
set var y4 $TCL(lindex $y3 0)
show var y4

###################################################################################################################
 This script demonstrates how to handle command prompts in the script
create vlan foo
delete vlan foo
SAVE config secondary
# handle prompt: Do you want to save configuration to secondary.cfg and overwrite it?    
# CLI.OUT system variable has response of the previous CLI command (SAVe config secondary)
# use regex to check if it is prompt, supply response if found
 regexp {.*overwrite it?.*} ${CLI.OUT} foundit
  IF ([info exists foundit]) THEN
   CLI yes
  ENDIF  

# handle second prompt: this prompt comes after you respond to first prompt.
 regexp {.*the default database?.*} ${CLI.OUT} foundit2
  IF ([info exists foundit2]) THEN
   CLI yes
 ENDIF

# Scripting supports two style of "if" statements : EXOS CLI style and tcl style

# EXOS CLI stype if statement (case sensitive : NO)
IF (!$match($isExos,true) ) THEN
 Echo "$deviceIP is an EXOS switch."
ELSE 
 Echo "$deviceIP is NOT an EXOS switch"
ENDIF

# tcl style if statement (case sensitive : YES)
if {[string equal $isExos "true"]} {
   Echo "$deviceIP is an EXOS switch."
else
   Echo "$deviceIP is NOT an EXOS switch."
}

set var CLI.OUT " "
CLI {show system | grep ignore-case "System MAC:"}
### quebra texto do resultado em linhas
set VAR_RESULT $TCL(split ${CLI.OUT} "\n")
### procura id (index) da linha onde tem a palavra chave "MAC:"
set VAR_LINE_NUMBER $TCL(lsearch -glob $VAR_RESULT *MAC:*)
### some o valor da linha + 1, pois o próprio comando tem o valor
set VAR_LINE_NUMBER [expr {$VAR_LINE_NUMBER + 1}]
### usa o numero da linha pra pegar o texto
set VAR_TEXT $TCL(lindex $VAR_RESULT $VAR_LINE_NUMBER)
### troca o delimitado da linha de texto para padrao split
set VAR_PREBLOCK [string map {"MAC:" "\;"} $VAR_TEXT]
### separa o text por delimitado ";"
set VAR_BLOCK [split $VAR_PREBLOCK ";"]
### pega valor na coluna 2
set VAR_MAC [lindex $VAR_BLOCK 1]
echo $VAR_MAC

set var CLI.OUT " "
CLI {show system | grep ignore-case "System MAC:"}
set VAR_RESULT $TCL(split ${CLI.OUT} "\n")
set VAR_LINE_NUMBER $TCL(lsearch -glob $VAR_RESULT *MAC:*)
set VAR_LINE_NUMBER [expr {$VAR_LINE_NUMBER + 1}]
set VAR_TEXT $TCL(lindex $VAR_RESULT $VAR_LINE_NUMBER)
set VAR_PREBLOCK [string map {"MAC:" "\;"} $VAR_TEXT]
set VAR_BLOCK [split $VAR_PREBLOCK ";"]
set VAR_MAC [lindex $VAR_BLOCK 1]
echo $VAR_MAC
